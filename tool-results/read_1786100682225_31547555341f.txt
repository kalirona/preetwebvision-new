    43→type AuthorOption = { name: string; initials: string; accent: string }

    45→// Unique authors derived from BLOG_POSTS (first occurrence wins for initials/accent).
    46→const AUTHORS: AuthorOption[] = (() => {
    47→  const seen = new Set<string>()
    48→  const out: AuthorOption[] = []
    49→  for (const p of BLOG_POSTS) {
    50→    if (seen.has(p.author)) continue
    51→    seen.add(p.author)
    52→    out.push({ name: p.author, initials: p.authorInitials, accent: p.authorAccent })
    53→  }
    54→  return out
    55→})()

    61→// Lowercased haystack of everything searchable on a post (title + excerpt + content).
    62→function postSearchText(post: BlogPost): string {
    63→  const contentText = post.content
    64→    .map((b) => (b.type === 'ul' ? (b.items ?? []).join(' ') : (b.text ?? '')))
    65→    .join(' ')
    66→  return `${post.title} ${post.excerpt} ${contentText}`.toLowerCase()
    67→}

   384→function ArticleView({ post, onBack, onOpen }: { post: BlogPost; onBack: () => void; onOpen: (p: BlogPost) => void }) {
   385→  const { setPage } = useNav()
   386→  const [progress, setProgress] = React.useState(0)
   387→  const articleRef = React.useRef<HTMLDivElement>(null)
   388→
   389→  // Reading progress + view tracking
   390→  React.useEffect(() => {
   391→    const onScroll = () => {
   392→      const el = articleRef.current
   393→      if (!el) return
   394→      const rect = el.getBoundingClientRect()
   395→      const total = el.scrollHeight - window.innerHeight
   396→      const scrolled = Math.min(Math.max(-rect.top, 0), total)
   397→      setProgress(total > 0 ? (scrolled / total) * 100 : 0)
   398→    }
   399→    window.addEventListener('scroll', onScroll, { passive: true })
   400→    onScroll()
   401→    // Track view (best-effort)
   402→    fetch('/api/blog', {
   403→      method: 'POST',
   404→      headers: { 'Content-Type': 'application/json' },
   405→      body: JSON.stringify({ slug: post.slug }),
   406→    }).catch(() => {})
   407→    return () => window.removeEventListener('scroll', onScroll)
   408→  }, [post.slug])
   409→
   410→  // Inject JSON-LD structured data for SEO (schema.org Article)
   411→  React.useEffect(() => {
   412→    const jsonLd = {
   413→      '@context': 'https://schema.org',
   414→      '@type': 'Article',
   415→      headline: post.title,
   416→      description: post.excerpt,
   417→      datePublished: post.date,
   418→      dateModified: post.date,
   419→      author: {
   420→        '@type': 'Person',
   421→        name: post.author,
   422→        jobTitle: post.authorRole,
   423→      },
   424→      publisher: {
   425→        '@type': 'Organization',
   426→        name: 'Preet Web Vision',
   427→        url: 'https://preetwebvision.com',
   428→      },
   429→      articleSection: post.category,
   430→      wordCount: post.content.reduce((acc, b) => acc + (b.text?.split(/\s+/).length ?? 0), 0),
   431→      keywords: post.category,
   432→    }
   433→    const script = document.createElement('script')
   434→    script.type = 'application/ld+json'
   435→    script.text = JSON.stringify(jsonLd)
   436→    script.dataset.dynamic = 'true'
   437→    document.head.appendChild(script)
   438→    return () => {
   439→      document.head.querySelectorAll('script[data-dynamic="true"]').forEach((s) => s.remove())
   440→    }
   441→  }, [post])
   442→
   443→  const related = BLOG_POSTS.filter((p) => p.id !== post.id && p.category === post.category).slice(0, 3)
   444→  const fallbackRelated = BLOG_POSTS.filter((p) => p.id !== post.id).slice(0, 3)
   445→  const relatedPosts = related.length >= 2 ? related : fallbackRelated
   446→
   447→  const share = async () => {
   448→    const url = typeof window !== 'undefined' ? window.location.href : ''
   449→    try {
   450→      if (navigator.share) {
   451→        await navigator.share({ title: post.title, url })
   452→      } else {
   453→        await navigator.clipboard.writeText(url)
   454→        toast.success('Link copied to clipboard!')
   455→      }
   456→    } catch {
   457→      /* user cancelled */
   458→    }
   459→  }
   460→
   461→  return (
   462→    <div className="relative">
   463→      {/* Reading progress bar */}
   464→      <div className="fixed inset-x-0 top-0 z-[55] h-1 bg-transparent">
   465→        <div
   466→          className="reading-progress h-full transition-[width] duration-150"
   467→          style={{ width: `${progress}%` }}
   468→        />
   469→      </div>
   470→
   471→      {/* Hero */}
   472→      <section className="relative overflow-hidden pb-10 pt-6">
   473→        <div className={cn('pointer-events-none absolute inset-0 bg-gradient-to-br opacity-10', post.gradient)} />
   474→        <div className="pointer-events-none absolute -top-24 left-1/2 size-96 -translate-x-1/2 rounded-full bg-brand-gradient opacity-15 blur-3xl" />
   475→        <div className="relative mx-auto max-w-3xl px-4 sm:px-6">
   476→          <nav className="breadcrumb" aria-label="Breadcrumb">
   477→            <button
   478→              onClick={() => setPage('home')}
   479→              className="hover:text-foreground"
   480→            >
   481→              Home
   482→            </button>
   483→            <ChevronRight className="size-3" />
   484→            <button
   485→              onClick={onBack}
   486→              className="group inline-flex items-center gap-1 font-medium hover:text-foreground"
   487→            >
   488→              <ArrowLeft className="size-3.5 transition-transform group-hover:-translate-x-0.5" />
   489→              Blog
   490→            </button>
   491→            <ChevronRight className="size-3" />
   492→            <span className="truncate text-foreground/70">{post.category}</span>
   493→          </nav>
   494→          <div className="mt-6 flex flex-wrap items-center gap-2 text-xs text-muted-foreground">
   495→            <span className="rounded-full bg-brand-gradient-soft px-2.5 py-1 font-semibold text-foreground">
   496→              {post.category}
   497→            </span>
   498→            <span className="flex items-center gap-1"><Calendar className="size-3" />{formatDate(post.date)}</span>
   499→            <span className="flex items-center gap-1"><Clock className="size-3" />{post.readingMinutes} min read</span>
   500→          </div>
   501→          <h1 className="mt-4 font-display text-3xl font-bold leading-[1.1] tracking-tight text-balance sm:text-4xl lg:text-5xl">

   655→/* ============ INLINE NEWSLETTER SIGNUP (in articles) ============ */
   656→function BlogNewsletterSignup() {
   657→  const [email, setEmail] = React.useState('')
   658→  const [status, setStatus] = React.useState<'idle' | 'loading' | 'success' | 'error'>('idle')
   659→
   660→  const submit = async (e: React.FormEvent) => {
   661→    e.preventDefault()
   662→    if (!email || status === 'loading') return
   663→    setStatus('loading')
   664→    try {
   665→      const res = await fetch('/api/newsletter', {
   666→        method: 'POST',
   667→        headers: { 'Content-Type': 'application/json' },
   668→        body: JSON.stringify({ email, source: 'blog-article' }),
   669→      })
   670→      const data = await res.json()
   671→      if (!res.ok || !data.ok) throw new Error(data.error || 'Failed')
   672→      setStatus('success')
   673→      setEmail('')
   674→      toast.success("You're on the list!", {
   675→        description: 'Expect growth tips and studio updates in your inbox.',
   676→      })
   677→    } catch {
   678→      setStatus('error')
   679→      toast.error('Could not subscribe. Please try again.')
   680→    }
   681→  }
   682→
   683→  return (
   684→    <div className="relative mt-12 overflow-hidden rounded-3xl border border-border/60 bg-muted/20 p-6 sm:p-8">
   685→      <div className="pointer-events-none absolute -right-12 -top-12 size-40 rounded-full bg-brand-gradient opacity-15 blur-3xl" />
   686→      <div className="pointer-events-none absolute inset-0 grid-bg opacity-20 [mask-image:radial-gradient(ellipse_at_right,#000,transparent_70%)]" />
   687→      <div className="relative">
   688→        <div className="flex items-start gap-4">
   689→          <span className="grid size-12 shrink-0 place-items-center rounded-2xl bg-brand-gradient text-white shadow-lg">
   690→            <Mail className="size-5" />
   691→          </span>
   692→          <div className="flex-1">
   693→            <h3 className="font-display text-lg font-bold">
   694→              Enjoyed this? Get the next one in your <span className="text-gradient-brand">inbox</span>.
   695→            </h3>
   696→            <p className="mt-1 text-sm text-muted-foreground">
   697→              Weekly insights on web, AI, SEO and growth. No spam, ever.
   698→            </p>
   699→          </div>
   700→        </div>
   701→        {status === 'success' ? (
   702→          <div className="mt-5 flex items-center gap-2 rounded-xl border border-emerald-500/30 bg-emerald-500/10 p-4 text-sm">
   703→            <CheckCircle2 className="size-5 shrink-0 text-emerald-500" />
   704→            <span>
   705→              <span className="font-semibold">You&apos;re subscribed!</span>{' '}
   706→              <span className="text-muted-foreground">Look out for our next article.</span>
   707→            </span>
   708→          </div>
   709→        ) : (
   710→          <form onSubmit={submit} className="mt-5 flex flex-col gap-2 sm:flex-row">
   711→            <Input
   712→              type="email"
   713→              required
   714→              value={email}
   715→              onChange={(e) => setEmail(e.target.value)}
   716→              placeholder="you@example.com"
   717→              className="h-11 flex-1 rounded-full bg-background/60"
   718→              disabled={status === 'loading'}
   719→            />
   720→            <Button
   721→              type="submit"
   722→              disabled={status === 'loading'}
   723→              className="h-11 shrink-0 rounded-full bg-brand-gradient text-white"
   724→            >
   725→              {status === 'loading' ? <Loader2 className="size-4 animate-spin" /> : <Send className="size-4" />}
   726→              {status === 'loading' ? 'Subscribing...' : 'Subscribe'}
   727→            </Button>
   728→          </form>
   729→        )}
   730→        <p className="mt-2 text-[11px] text-muted-foreground">
   731→          Join 2,000+ founders and operators. Unsubscribe anytime.
   732→        </p>
   733→      </div>
   734→    </div>

   739→export function BlogPage() {
   740→  const [activePost, setActivePost] = React.useState<BlogPost | null>(null)
   741→
   742→  const open = React.useCallback((post: BlogPost) => {
   743→    setActivePost(post)
   744→    if (typeof window !== 'undefined') window.scrollTo({ top: 0, behavior: 'instant' as ScrollBehavior })
   745→  }, [])
   746→
   747→  const back = React.useCallback(() => {
   748→    setActivePost(null)
   749→    if (typeof window !== 'undefined') window.scrollTo({ top: 0, behavior: 'instant' as ScrollBehavior })
   750→  }, [])
   751→
   752→  // Listen for command palette "open blog post" events
   753→  React.useEffect(() => {
   754→    const onOpenPost = (e: Event) => {
   755→      const slug = (e as CustomEvent<string>).detail
   756→      const post = BLOG_POSTS.find((p) => p.slug === slug)
   757→      if (post) open(post)
   758→    }
   759→    window.addEventListener('open-blog-post', onOpenPost as EventListener)
   760→    return () => window.removeEventListener('open-blog-post', onOpenPost as EventListener)
   761→  }, [open])
   762→
   763→  return (
   764→    <div className="relative">
   765→      <AnimatePresence mode="wait">
   766→        {activePost ? (
   767→          <motion.div
   768→            key={activePost.id}
   769→            initial={{ opacity: 0 }}
   770→            animate={{ opacity: 1 }}
   771→            exit={{ opacity: 0 }}
   772→            transition={{ duration: 0.3 }}
   773→          >
   774→            <ArticleView post={activePost} onBack={back} onOpen={open} />
   775→          </motion.div>
   776→        ) : (
   777→          <motion.div
   778→            key="grid"
   779→            initial={{ opacity: 0 }}
   780→            animate={{ opacity: 1 }}
   781→            exit={{ opacity: 0 }}
   782→            transition={{ duration: 0.3 }}
   783→          >
   784→            <BlogHero />
   785→            <BlogGrid onOpen={open} />
   786→            <BlogCta />
   787→          </motion.div>
   788→        )}
   789→      </AnimatePresence>
   790→    </div>
   791→  )
   792→}
