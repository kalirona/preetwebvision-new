   244→
   245→      {/* featured ambient glow */}

   266→              Plan

   268→          )}
   269→        </div>

   285→                featured && 'text-gradient-brand'
   286→              )}
   287→            >
   288→              {showPrice}
   289→            </motion.span>

   316→            </Button>

   319→              onClick={onCta}
   320→              variant="outline"
