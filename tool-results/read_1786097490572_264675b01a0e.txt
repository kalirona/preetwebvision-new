    39→type AuthorOption = { name: string; initials: string; accent: string }

    41→// Unique authors derived from BLOG_POSTS (first occurrence wins for initials/accent).
    42→const AUTHORS: AuthorOption[] = (() => {
    43→  const seen = new Set<string>()
    44→  const out: AuthorOption[] = []
    45→  for (const p of BLOG_POSTS) {
    46→    if (seen.has(p.author)) continue
    47→    seen.add(p.author)
    48→    out.push({ name: p.author, initials: p.authorInitials, accent: p.authorAccent })
    49→  }
    50→  return out
    51→})()

    53→function formatDate(iso: string) {

    57→// Lowercased haystack of everything searchable on a post (title + excerpt + content).
    58→function postSearchText(post: BlogPost): string {
    59→  const contentText = post.content
    60→    .map((b) => (b.type === 'ul' ? (b.items ?? []).join(' ') : (b.text ?? '')))
    61→    .join(' ')
    62→  return `${post.title} ${post.excerpt} ${contentText}`.toLowerCase()
    63→}

   356→/* ============ ARTICLE VIEW ============ */
   357→function ArticleView({ post, onBack, onOpen }: { post: BlogPost; onBack: () => void; onOpen: (p: BlogPost) => void }) {
   358→  const { setPage } = useNav()
   359→  const [progress, setProgress] = React.useState(0)
   360→  const articleRef = React.useRef<HTMLDivElement>(null)
   361→
   362→  // Reading progress + view tracking
   363→  React.useEffect(() => {
   364→    const onScroll = () => {
   365→      const el = articleRef.current
   366→      if (!el) return
   367→      const rect = el.getBoundingClientRect()
   368→      const total = el.scrollHeight - window.innerHeight
   369→      const scrolled = Math.min(Math.max(-rect.top, 0), total)
   370→      setProgress(total > 0 ? (scrolled / total) * 100 : 0)
   371→    }
   372→    window.addEventListener('scroll', onScroll, { passive: true })
   373→    onScroll()
   374→    // Track view (best-effort)
   375→    fetch('/api/blog', {
   376→      method: 'POST',
   377→      headers: { 'Content-Type': 'application/json' },
   378→      body: JSON.stringify({ slug: post.slug }),
   379→    }).catch(() => {})
   380→    return () => window.removeEventListener('scroll', onScroll)
   381→  }, [post.slug])
   382→
   383→  const related = BLOG_POSTS.filter((p) => p.id !== post.id && p.category === post.category).slice(0, 3)
   384→  const fallbackRelated = BLOG_POSTS.filter((p) => p.id !== post.id).slice(0, 3)
   385→  const relatedPosts = related.length >= 2 ? related : fallbackRelated
   386→
   387→  const share = async () => {
   388→    const url = typeof window !== 'undefined' ? window.location.href : ''
   389→    try {
   390→      if (navigator.share) {
   391→        await navigator.share({ title: post.title, url })
   392→      } else {
   393→        await navigator.clipboard.writeText(url)
   394→        toast.success('Link copied to clipboard!')
   395→      }
   396→    } catch {
   397→      /* user cancelled */
   398→    }
   399→  }
   400→
   401→  return (
   402→    <div className="relative">
   403→      {/* Reading progress bar */}
   404→      <div className="fixed inset-x-0 top-0 z-[55] h-1 bg-transparent">
   405→        <div
   406→          className="reading-progress h-full transition-[width] duration-150"
   407→          style={{ width: `${progress}%` }}
   408→        />
   409→      </div>
   410→
   411→      {/* Hero */}
   412→      <section className="relative overflow-hidden pb-10 pt-6">
   413→        <div className={cn('pointer-events-none absolute inset-0 bg-gradient-to-br opacity-10', post.gradient)} />
   414→        <div className="pointer-events-none absolute -top-24 left-1/2 size-96 -translate-x-1/2 rounded-full bg-brand-gradient opacity-15 blur-3xl" />
   415→        <div className="relative mx-auto max-w-3xl px-4 sm:px-6">
   416→          <button
   417→            onClick={onBack}
   418→            className="group inline-flex items-center gap-1.5 text-sm font-medium text-muted-foreground transition-colors hover:text-foreground"
   419→          >
   420→            <ArrowLeft className="size-4 transition-transform group-hover:-translate-x-0.5" />
   421→            All articles
   422→          </button>
   423→          <div className="mt-6 flex flex-wrap items-center gap-2 text-xs text-muted-foreground">
   424→            <span className="rounded-full bg-brand-gradient-soft px-2.5 py-1 font-semibold text-foreground">
   425→              {post.category}
   426→            </span>
   427→            <span className="flex items-center gap-1"><Calendar className="size-3" />{formatDate(post.date)}</span>
   428→            <span className="flex items-center gap-1"><Clock className="size-3" />{post.readingMinutes} min read</span>
   429→          </div>
   430→          <h1 className="mt-4 font-display text-3xl font-bold leading-[1.1] tracking-tight text-balance sm:text-4xl lg:text-5xl">
   431→            {post.title}
   432→          </h1>
   433→          <p className="mt-4 text-lg leading-relaxed text-muted-foreground">{post.excerpt}</p>
   434→
   435→          {/* Author + share */}
   436→          <div className="mt-7 flex items-center justify-between">
   437→            <div className="flex items-center gap-3">
   438→              <Avatar className="size-11 border-2 border-background">
   439→                <AvatarFallback className={cn('bg-gradient-to-br text-white', post.authorAccent)}>
   440→                  {post.authorInitials}
   441→                </AvatarFallback>
   442→              </Avatar>
   443→              <div>
   444→                <p className="font-display text-sm font-bold">{post.author}</p>
   445→                <p className="text-xs text-muted-foreground">{post.authorRole}</p>
   446→              </div>
   447→            </div>
   448→            <button
   449→              onClick={share}
   450→              className="flex items-center gap-1.5 rounded-full border border-border/70 bg-muted/30 px-3.5 py-2 text-xs font-semibold transition-colors hover:text-foreground hover:border-border"
   451→            >
   452→              <Share2 className="size-3.5" />
   453→              Share
   454→            </button>
   455→          </div>
   456→        </div>
   457→      </section>
   458→
   459→      {/* Cover */}
   460→      <div className="mx-auto max-w-4xl px-4 sm:px-6">
   461→        <div className={cn('relative h-56 overflow-hidden rounded-3xl bg-gradient-to-br sm:h-72', post.gradient)}>
   462→          <div className="absolute inset-0 grid-bg opacity-20" />
   463→          <div className="absolute inset-0 bg-gradient-to-t from-black/40 to-transparent" />
   464→          <span className="absolute left-6 top-6 text-6xl drop-shadow-lg">{post.emoji}</span>
   465→        </div>
   466→      </div>
   467→
   468→      {/* Body */}
   469→      <article ref={articleRef} className="mx-auto max-w-3xl px-4 py-12 sm:px-6">
   470→        <div className="prose-brand">
   471→          {post.content.map((block, i) => {
   472→            if (block.type === 'h2') return <h2 key={i}>{block.text}</h2>
   473→            if (block.type === 'h3') return <h3 key={i}>{block.text}</h3>
   474→            if (block.type === 'quote')
   475→              return <blockquote key={i}>{block.text}</blockquote>
   476→            if (block.type === 'ul')
   477→              return (
   478→                <ul key={i}>
   479→                  {block.items?.map((it, j) => <li key={j}>{it}</li>)}
   480→                </ul>
   481→              )
   482→            return <p key={i}>{block.text}</p>
   483→          })}
   484→        </div>
   485→

   538→/* ============ PAGE ============ */
   539→export function BlogPage() {
   540→  const [activePost, setActivePost] = React.useState<BlogPost | null>(null)
   541→
   542→  const open = React.useCallback((post: BlogPost) => {
   543→    setActivePost(post)
   544→    if (typeof window !== 'undefined') window.scrollTo({ top: 0, behavior: 'instant' as ScrollBehavior })
   545→  }, [])
   546→
   547→  const back = React.useCallback(() => {
   548→    setActivePost(null)
   549→    if (typeof window !== 'undefined') window.scrollTo({ top: 0, behavior: 'instant' as ScrollBehavior })
   550→  }, [])
   551→
   552→  // Listen for command palette "open blog post" events
   553→  React.useEffect(() => {
   554→    const onOpenPost = (e: Event) => {
   555→      const slug = (e as CustomEvent<string>).detail
   556→      const post = BLOG_POSTS.find((p) => p.slug === slug)
   557→      if (post) open(post)
   558→    }
   559→    window.addEventListener('open-blog-post', onOpenPost as EventListener)
   560→    return () => window.removeEventListener('open-blog-post', onOpenPost as EventListener)
   561→  }, [open])
   562→
   563→  return (
   564→    <div className="relative">
   565→      <AnimatePresence mode="wait">
   566→        {activePost ? (
   567→          <motion.div
   568→            key={activePost.id}
   569→            initial={{ opacity: 0 }}
   570→            animate={{ opacity: 1 }}
   571→            exit={{ opacity: 0 }}
   572→            transition={{ duration: 0.3 }}
   573→          >
   574→            <ArticleView post={activePost} onBack={back} onOpen={open} />
   575→          </motion.div>
   576→        ) : (
   577→          <motion.div
   578→            key="grid"
   579→            initial={{ opacity: 0 }}
   580→            animate={{ opacity: 1 }}
   581→            exit={{ opacity: 0 }}
   582→            transition={{ duration: 0.3 }}
   583→          >
   584→            <BlogHero />
   585→            <BlogGrid onOpen={open} />
   586→            <BlogCta />
   587→          </motion.div>
   588→        )}
   589→      </AnimatePresence>
   590→    </div>
   591→  )
   592→}
