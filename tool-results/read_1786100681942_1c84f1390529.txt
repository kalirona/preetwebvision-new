   249→/* ============================== PROJECT GALLERY ============================== */
   250→function ProjectGallery({ onOpen }: { onOpen: (p: Project) => void }) {
   251→  const categories: FilterId[] = React.useMemo(() => {
   252→    const unique = Array.from(new Set(PROJECTS.map((p) => p.category)))
   253→    return ['All', ...unique]
   254→  }, [])
   255→
   256→  const counts = React.useMemo(() => {
   257→    const c: Record<string, number> = { All: PROJECTS.length }
   258→    for (const p of PROJECTS) c[p.category] = (c[p.category] ?? 0) + 1
   259→    return c
   260→  }, [])
   261→
   262→  const [active, setActive] = React.useState<FilterId>('All')
   263→
   264→  const filtered = React.useMemo(
   265→    () => (active === 'All' ? PROJECTS : PROJECTS.filter((p) => p.category === active)),
   266→    [active]
   267→  )
   268→
   269→  return (
   270→    <section id="gallery" className="relative py-16 sm:py-24">
   271→      <div className="mx-auto max-w-7xl px-4 sm:px-6">
   272→        <SectionHeading
   273→          align="left"
   274→          eyebrow="The work"
   275→          title={
   276→            <>
   277→              Projects that <GradientText>ship outcomes</GradientText>
   278→            </>
   279→          }
   280→          description="Filter by discipline to explore how we&apos;ve helped brands grow with design, engineering, AI and growth."
   281→          className="max-w-2xl"
   282→        />
   283→
   284→        <div className="mt-10">
   285→          <FilterBar
   286→            active={active}
   287→            onChange={setActive}
   288→            categories={categories}
   289→            counts={counts}
   290→          />
   291→        </div>
   292→
   293→        {/* Grid with AnimatePresence + layout for smooth reflow */}
   294→        <motion.div
   295→          layout
   296→          className="mt-8 grid grid-cols-1 gap-5 sm:grid-cols-2 lg:grid-cols-3"
   297→        >
   298→          <AnimatePresence mode="popLayout">
   299→            {filtered.map((p) => (
   300→              <ProjectCard key={p.id} project={p} onOpen={onOpen} />
   301→            ))}
   302→          </AnimatePresence>
   303→        </motion.div>
   304→
   305→        {/* Empty state — defensive */}
   306→        {filtered.length === 0 && (
   307→          <div className="mt-10 rounded-3xl border border-dashed border-border/60 p-12 text-center text-muted-foreground">
   308→            No projects in this category yet.
   309→          </div>
   310→        )}
   311→
   312→        <Reveal delay={0.1}>
   313→          <div className="mt-12 flex flex-col items-center justify-between gap-4 rounded-3xl border border-border/60 bg-card/60 p-6 sm:flex-row sm:p-8">
   314→            <div>
   315→              <p className="font-display text-lg font-bold">Don&apos;t see your industry?</p>
   316→              <p className="mt-1 text-sm text-muted-foreground">
   317→                We&apos;ve shipped 180+ projects across nearly every vertical. Let&apos;s talk about yours.
   318→              </p>
   319→            </div>
   320→            <StartProjectButton />
   321→          </div>
   322→        </Reveal>
   323→      </div>
   324→    </section>
   325→  )
   326→}

   619→/* ============================== PAGE ============================== */
   620→export function PortfolioPage() {
   621→  const [active, setActive] = React.useState<Project | null>(null)
   622→
   623→  // Listen for "open-case-study" events from related-project links in the modal
   624→  React.useEffect(() => {
   625→    const onOpen = (e: Event) => {
   626→      const id = (e as CustomEvent<string>).detail
   627→      const proj = PROJECTS.find((p) => p.id === id)
   628→      if (proj) {
   629→        setActive(null)
   630→        setTimeout(() => setActive(proj), 50)
   631→      }
   632→    }
   633→    window.addEventListener('open-case-study', onOpen as EventListener)
   634→    return () => window.removeEventListener('open-case-study', onOpen as EventListener)
   635→  }, [])
   636→
   637→  return (
   638→    <div className="relative">
   639→      <Hero />
   640→      <ProjectGallery onOpen={setActive} />
   641→      <ProcessStrip />
   642→      <TrustedBy />
   643→      <TestimonialSection />
   644→      <FinalCta />
   645→      <CaseStudyModal project={active} onClose={() => setActive(null)} />
   646→    </div>
   647→  )
   648→}
