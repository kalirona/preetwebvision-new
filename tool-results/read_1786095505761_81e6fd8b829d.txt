   209→function PricingCard({
   210→  plan,
   211→  billing,
   212→  onCta,
   213→}: {
   214→  plan: PricingPlan
   215→  billing: BillingMode
   216→  onCta: () => void
   217→}) {
   218→  const featured = !!plan.featured
   219→  const monthly = MONTHLY_PRICING[plan.name]
   220→  const showPrice = billing === 'monthly' && monthly ? monthly.price : plan.price
   221→  const showPeriod = billing === 'monthly' && monthly ? monthly.period : plan.period
   222→  // For the Launch plan, monthly retainer isn't offered — show a note instead
   223→  const launchMonthlyNote = billing === 'monthly' && plan.name === 'Launch'
   224→  return (
   225→    <div
   226→      className={cn(
   227→        'group card-sheen relative flex h-full flex-col overflow-hidden rounded-3xl transition-all duration-300',
   228→        featured
   229→          ? 'gradient-border glow-brand lg:scale-105 lg:-translate-y-2'
   230→          : 'border border-border/60 bg-card hover:-translate-y-1 hover:border-border hover:shadow-xl'
   231→      )}
   232→    >
   233→      {/* featured gradient header strip */}
   234→      {featured && (
   235→        <div
   236→          className={cn(
   237→            'h-1.5 w-full bg-gradient-to-r',
   238→            plan.accent
   239→          )}
   240→          aria-hidden
   241→        />
   242→      )}
   243→
   244→      {/* featured ambient glow */}
   245→      {featured && (
   246→        <div
   247→          className="pointer-events-none absolute -top-20 left-1/2 size-64 -translate-x-1/2 rounded-full bg-brand-gradient opacity-20 blur-3xl"
   248→          aria-hidden
   249→        />
   250→      )}
   251→
   252→      <div className="relative flex flex-1 flex-col p-7 sm:p-8">
   253→        {/* Plan header */}
   254→        <div className="flex items-center justify-between">
   255→          <h3 className="font-display text-xl font-bold tracking-tight">
   256→            {plan.name}
   257→          </h3>
   258→          {featured ? (
   259→            <Badge className="rounded-full border-0 bg-brand-gradient px-3 py-1 text-xs font-semibold text-white shadow-md">
   260→              <Sparkles className="mr-1 size-3.5" />
   261→              Most popular
   262→            </Badge>
   263→          ) : (
   264→            <span className="rounded-full border border-border/60 bg-muted/40 px-3 py-1 text-xs font-medium text-muted-foreground">
   265→              Plan
   266→            </span>
   267→          )}
   268→        </div>
   269→
   270→        {/* Price */}
   271→        <div className="mt-5 flex items-end gap-1.5">
   272→          {launchMonthlyNote ? (
   273→            <span className="font-display text-3xl font-bold tracking-tight text-muted-foreground">
   274→              Retainer ready
   275→            </span>
   276→          ) : (
   277→            <motion.span
   278→              key={showPrice}
   279→              initial={{ opacity: 0, y: 8 }}
   280→              animate={{ opacity: 1, y: 0 }}
   281→              transition={{ duration: 0.3 }}
   282→              className={cn(
   283→                'font-display text-5xl font-bold tracking-tight',
   284→                featured && 'text-gradient-brand'
   285→              )}
   286→            >
   287→              {showPrice}
   288→            </motion.span>
   289→          )}
   290→          {!launchMonthlyNote && (
   291→            <span className="mb-1.5 text-sm font-medium text-muted-foreground">
   292→              {showPeriod}
   293→            </span>
   294→          )}
   295→        </div>
   296→
   297→        <p className="mt-3 text-sm leading-relaxed text-muted-foreground">
   298→          {launchMonthlyNote
   299→            ? 'Launch is a one-time project. Upgrade to a Growth or Enterprise retainer for ongoing work.'
   300→            : plan.description}
   301→        </p>
   302→
   303→        {/* CTA */}
   304→        <div className="mt-6">
   305→          {featured ? (
   306→            <Button
   307→              onClick={onCta}
   308→              className="group/cta relative h-11 w-full overflow-hidden rounded-full bg-brand-gradient text-white shadow-[0_8px_30px_-8px_rgba(255,45,117,0.6)] hover:shadow-[0_12px_40px_-8px_rgba(255,45,117,0.85)]"
   309→            >
   310→              <span className="relative z-10 flex items-center justify-center gap-2">
   311→                {plan.cta}
   312→                <ArrowRight className="size-4 transition-transform group-hover/cta:translate-x-1" />
   313→              </span>
   314→              <span className="absolute inset-0 animate-shimmer opacity-40" />
   315→            </Button>
   316→          ) : (
   317→            <Button
   318→              onClick={onCta}
   319→              variant="outline"
   320→              className="h-11 w-full rounded-full"
   321→            >
   322→              {plan.cta}
   323→              <ArrowRight className="size-4" />
   324→            </Button>
   325→          )}
   326→        </div>
   327→
   328→        {/* Divider */}
   329→        <div className="my-7 h-px w-full bg-border/60" />
   330→
   331→        {/* Features */}
   332→        <ul className="flex flex-1 flex-col gap-3.5">
   333→          {plan.features.map((f) => (
   334→            <li key={f} className="flex items-start gap-3 text-sm">
   335→              <span
   336→                className={cn(
   337→                  'mt-0.5 grid size-5 shrink-0 place-items-center rounded-full bg-gradient-to-br text-white shadow-sm',
   338→                  plan.accent
   339→                )}
   340→              >
   341→                <Check className="size-3.5" strokeWidth={3} />
   342→              </span>
   343→              <span className="text-foreground/90">{f}</span>
   344→            </li>
   345→          ))}
   346→        </ul>
   347→      </div>
   348→    </div>
   349→  )
   350→}
