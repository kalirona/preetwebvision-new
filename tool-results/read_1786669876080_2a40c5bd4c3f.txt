    41→/* ============ Notification Bell ============ */
    42→function NotificationBell() {
    43→  const [open, setOpen] = React.useState(false)
    44→  const [notifications, setNotifications] = React.useState<Array<{ id: string; type: string; title: string; message: string; read: boolean; createdAt: string }>>([])
    45→  const [unread, setUnread] = React.useState(0)
    46→  const ref = React.useRef<HTMLDivElement>(null)
    47→
    48→  const fetchNotifs = React.useCallback(async () => {
    49→    try {
    50→      const res = await fetch('/api/admin/notifications')
    51→      const data = await res.json()
    52→      setNotifications(data.notifications || [])
    53→      setUnread(data.unreadCount || 0)
    54→    } catch { /* ignore */ }
    55→  }, [])
    56→
    57→  React.useEffect(() => {
    58→    fetchNotifs()
    59→    const id = setInterval(fetchNotifs, 15000)
    60→    return () => clearInterval(id)
    61→  }, [fetchNotifs])
    62→
    63→  React.useEffect(() => {
    64→    const onClick = (e: MouseEvent) => {
    65→      if (ref.current && !ref.current.contains(e.target as Node)) setOpen(false)
    66→    }
    67→    document.addEventListener('mousedown', onClick)
    68→    return () => document.removeEventListener('mousedown', onClick)
    69→  }, [])
    70→
    71→  const markAllRead = async () => {
    72→    await fetch('/api/admin/notifications', { method: 'PATCH', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ markAllRead: true }) })
    73→    setNotifications((prev) => prev.map((n) => ({ ...n, read: true })))
    74→    setUnread(0)
    75→  }
    76→
    77→  return (
    78→    <div ref={ref} className="relative">
    79→      <button
    80→        onClick={() => setOpen((o) => !o)}
    81→        className="relative grid size-9 place-items-center rounded-full border border-border/70 bg-muted/30 text-muted-foreground transition-colors hover:text-foreground"
    82→        aria-label="Notifications"
    83→      >
    84→        <Bell className="size-4" />
    85→        {unread > 0 && (
    86→          <span className="absolute -right-1 -top-1 grid min-w-[18px] h-[18px] place-items-center rounded-full bg-brand-gradient px-1 text-[10px] font-bold text-white">
    87→            {unread > 9 ? '9+' : unread}
    88→          </span>
    89→        )}
    90→      </button>
    91→      <AnimatePresence>
    92→        {open && (
    93→          <motion.div
    94→            initial={{ opacity: 0, y: 10, scale: 0.97 }}
    95→            animate={{ opacity: 1, y: 0, scale: 1 }}
    96→            exit={{ opacity: 0, y: 10, scale: 0.97 }}
    97→            className="absolute right-0 top-full z-50 mt-2 w-80 overflow-hidden rounded-2xl border border-border/60 bg-card shadow-2xl"
    98→          >
    99→            <div className="flex items-center justify-between border-b border-border/60 p-3">
   100→              <p className="text-sm font-bold">Notifications</p>
   101→              {unread > 0 && (
   102→                <button onClick={markAllRead} className="text-xs font-semibold text-[var(--brand-pink)] hover:underline">
   103→                  Mark all read
   104→                </button>
   105→              )}
   106→            </div>
   107→            <div className="max-h-80 overflow-y-auto">
   108→              {notifications.length === 0 ? (
   109→                <p className="p-6 text-center text-sm text-muted-foreground">No notifications yet.</p>
   110→              ) : (
   111→                notifications.slice(0, 10).map((n) => (
   112→                  <a key={n.id} href={n.link || '/admin/dashboard'} className={cn('block border-b border-border/40 p-3 transition-colors hover:bg-muted/30', !n.read && 'bg-brand-gradient-soft')}>
   113→                    <div className="flex items-start gap-2">
   114→                      <span className={cn('mt-1 size-2 shrink-0 rounded-full', n.read ? 'bg-muted-foreground/30' : 'bg-[var(--brand-pink)]')} />
   115→                      <div className="min-w-0 flex-1">
   116→                        <p className="text-sm font-semibold">{n.title}</p>
   117→                        <p className="truncate text-xs text-muted-foreground">{n.message}</p>
   118→                        <p className="mt-0.5 text-[10px] text-muted-foreground">{timeAgo(n.createdAt)}</p>
   119→                      </div>
   120→                    </div>
   121→                  </a>
   122→                ))
   123→              )}
   124→            </div>
   125→          </motion.div>
   126→        )}
   127→      </AnimatePresence>
   128→    </div>
   129→  )
   130→}

   165→function timeAgo(iso: string) {
   166→  const diff = Date.now() - new Date(iso).getTime()
   167→  const mins = Math.floor(diff / 60000)
   168→  if (mins < 1) return 'just now'
   169→  if (mins < 60) return `${mins}m ago`
   170→  const hrs = Math.floor(mins / 60)
   171→  if (hrs < 24) return `${hrs}h ago`
   172→  return `${Math.floor(hrs / 24)}d ago`
   173→}

   343→function SubmissionsTab() {
   344→  const [submissions, setSubmissions] = React.useState<Submission[]>([])
   345→  const [loading, setLoading] = React.useState(true)
   346→  const [selected, setSelected] = React.useState<Submission | null>(null)
   347→  const [filter, setFilter] = React.useState('all')
   348→  const [search, setSearch] = React.useState('')
   349→
   350→  const fetchSubs = React.useCallback(async () => {
   351→    setLoading(true)
   352→    try {
   353→      const res = await fetch(`/api/admin/submissions?status=${filter}`)
   354→      const data = await res.json()
   355→      setSubmissions(data.submissions || [])
   356→    } catch { /* ignore */ } finally { setLoading(false) }
   357→  }, [filter])
   358→
   359→  React.useEffect(() => { fetchSubs() }, [fetchSubs])
   360→
   361→  const updateStatus = async (id: string, status: string) => {
   362→    await fetch('/api/admin/submissions', {
   363→      method: 'PATCH',
   364→      headers: { 'Content-Type': 'application/json' },
   365→      body: JSON.stringify({ id, status }),
   366→    })
   367→    setSubmissions((prev) => prev.map((s) => (s.id === id ? { ...s, status } : s)))
   368→    if (selected?.id === id) setSelected((s) => (s ? { ...s, status } : s))
   369→  }
   370→
   371→  const filtered = submissions.filter((s) => {
   372→    if (search) {
   373→      const q = search.toLowerCase()
   374→      return s.name.toLowerCase().includes(q) || s.email.toLowerCase().includes(q) || s.message.toLowerCase().includes(q)
   375→    }
   376→    return true
   377→  })
   378→
   379→  return (
   380→    <div className="grid gap-0 lg:grid-cols-[420px_1fr]">
   381→      {/* List */}
   382→      <div className="border-r border-border/60">
   383→        <div className="space-y-3 border-b border-border/60 p-4">

   388→          <div className="flex gap-1.5">
   389→            {['all', 'new', 'read', 'replied', 'archived'].map((f) => (
   390→              <button key={f} onClick={() => setFilter(f)} className={cn('rounded-full px-3 py-1 text-xs font-semibold capitalize transition-colors', filter === f ? 'bg-brand-gradient text-white' : 'bg-muted/40 text-muted-foreground hover:text-foreground')}>
   391→                {f}
   392→              </button>
   393→            ))}
   394→          </div>
   395→        </div>
   396→        <div className="max-h-[60vh] overflow-y-auto lg:max-h-[calc(100vh-280px)]">
   397→          {loading ? (
   398→            <div className="flex justify-center py-12"><Loader2 className="size-5 animate-spin text-muted-foreground" /></div>
   399→          ) : filtered.length === 0 ? (
   400→            <div className="flex flex-col items-center py-12 text-center">
   401→              <Mail className="size-8 text-muted-foreground/40" />
   402→              <p className="mt-2 text-sm text-muted-foreground">No submissions found.</p>
   403→            </div>
   404→          ) : (

   407→                <div className="flex items-center justify-between">

'bg-emerald-500/15 text-emerald-600 dark:text-emerald-400' : sub.status === 'replied' ? 'bg-amber-500/15 text-amber-600 dark:text-amber-400' : 'bg-muted text-muted-foreground')}>{sub.status}</span>

   442→            </div>
   443→            <div className="rounded-2xl border border-border/60 bg-card p-5">
   444→              <p className="mb-2 text-xs font-semibold uppercase tracking-wide text-muted-foreground">Message</p>

   447→            <div className="flex gap-2">
   448→              <a href={`mailto:${selected.email}?subject=Re: Your project inquiry&body=Hi ${selected.name},%0D%0A%0D%0AThanks for reaching out to Preet Web Vision! %0D%0A%0D%0A`}>

   812→function AffiliateForm({ initial, onClose, onSaved }: { initial: AffiliateRow | null; onClose: () => void; onSaved: () => void }) {

   823→  const handleSubmit = async (e: React.FormEvent) => {
   824→    e.preventDefault()
   825→    if (!title.trim() || !affiliateUrl.trim()) {

  1265→      const res = await fetch('/api/admin/blog')
  1266→      const data = await res.json()
  1267→      setPosts(data.posts || [])
  1268→    } catch { /* ignore */ } finally { setLoading(false) }
  1269→  }, [])
  1270→
  1271→  React.useEffect(() => { fetchPosts() }, [fetchPosts])
  1272→
  1273→  const handleDelete = async (id: string) => {

  1276→    setPosts((prev) => prev.filter((p) => p.id !== id))

  1278→  }

  1280→  const handleToggleFeatured = async (post: BlogPostRow) => {
  1281→    await fetch('/api/admin/blog', {
  1282→      method: 'PUT',
  1283→      headers: { 'Content-Type': 'application/json' },
  1284→      body: JSON.stringify({ id: post.id, featured: !post.featured }),
  1285→    })
  1286→    setPosts((prev) => prev.map((p) => (p.id === post.id ? { ...p, featured: !p.featured } : p)))
  1287→  }
  1288→
  1289→  const handleToggleStatus = async (post: BlogPostRow) => {
  1290→    const newStatus = post.status === 'published' ? 'draft' : 'published'
  1291→    await fetch('/api/admin/blog', {
  1292→      method: 'PUT',
  1293→      headers: { 'Content-Type': 'application/json' },
  1294→      body: JSON.stringify({ id: post.id, status: newStatus }),
  1295→    })
  1296→    setPosts((prev) => prev.map((p) => (p.id === post.id ? { ...p, status: newStatus } : p)))
  1297→  }
  1298→
  1299→  return (
  1300→    <div className="space-y-6">
  1301→      <div className="flex items-center justify-between">

  1303→          <h2 className="font-display text-2xl font-bold tracking-tight">Blog Posts</h2>
  1304→          <p className="mt-1 text-sm text-muted-foreground">Create, edit, and publish blog articles. Changes appear instantly on the website.</p>
  1305→        </div>

  1307→          <BookOpen className="size-4" />
  1308→          New post
  1309→        </Button>
  1310→      </div>
  1311→
  1312→      {showForm && (
  1313→        <BlogPostForm
  1314→          initial={editing}
  1315→          onClose={() => { setShowForm(false); setEditing(null) }}
  1316→          onSaved={() => { setShowForm(false); setEditing(null); fetchPosts() }}
  1317→        />
  1318→      )}
  1319→
  1320→      {loading ? (
  1321→        <div className="flex justify-center py-12"><Loader2 className="size-5 animate-spin text-muted-foreground" /></div>
  1322→      ) : posts.length === 0 ? (
  1323→        <div className="flex flex-col items-center rounded-2xl border border-dashed border-border/60 py-16 text-center">
  1324→          <BookOpen className="size-10 text-muted-foreground/40" />
  1325→          <p className="mt-3 font-display text-lg font-bold">No blog posts yet</p>
  1326→          <p className="mt-1 text-sm text-muted-foreground">Write your first article to publish it on the blog.</p>
  1327→          <Button onClick={() => { setEditing(null); setShowForm(true) }} className="mt-4 rounded-full bg-brand-gradient text-white">
  1328→            Write first post
  1329→          </Button>
  1330→        </div>
  1331→      ) : (
  1332→        <div className="space-y-3">
  1333→          {posts.map((post) => (
  1334→            <div key={post.id} className="flex items-start gap-4 rounded-2xl border border-border/60 bg-card p-4">
  1335→              <div className="size-16 shrink-0 overflow-hidden rounded-xl bg-muted">
  1336→                {post.imageUrl && <img src={post.imageUrl} alt={post.title} className="size-full object-cover" />}
  1337→              </div>
  1338→              <div className="min-w-0 flex-1">
  1339→                <div className="flex items-start justify-between gap-2">
  1340→                  <div className="min-w-0">
  1341→                    <p className="truncate font-display text-sm font-bold">{post.title}</p>
  1342→                    <p className="truncate text-xs text-muted-foreground">{post.excerpt || 'No excerpt'}</p>
  1343→                  </div>
  1344→                  <div className="flex shrink-0 gap-1">
  1345→                    <button onClick={() => handleToggleFeatured(post)} className={cn('rounded-full px-2 py-1 text-[10px] font-bold uppercase', post.featured ? 'bg-amber-500/15 text-amber-600 dark:text-amber-400' : 'bg-muted text-muted-foreground')} title="Toggle featured">
  1346→                      ★
  1347→                    </button>
  1348→                    <button onClick={() => handleToggleStatus(post)} className={cn('rounded-full px-2 py-1 text-[10px] font-bold uppercase', post.status === 'published' ? 'bg-emerald-500/15 text-emerald-600 dark:text-emerald-400' : 'bg-muted text-muted-foreground')}>
  1349→                      {post.status}
  1350→                    </button>
  1351→                  </div>
  1352→                </div>
  1353→                <div className="mt-2 flex items-center gap-3 text-[10px] text-muted-foreground">
  1354→                  <span>{post.category}</span>
  1355→                  <span>·</span>
  1356→                  <span>{post.author}</span>
  1357→                  <span>·</span>
  1358→                  <span>{new Date(post.createdAt).toLocaleDateString()}</span>
  1359→                </div>
  1360→                <div className="mt-2 flex gap-2">
  1361→                  <Button size="sm" variant="ghost" onClick={() => { setEditing(post); setShowForm(true) }} className="rounded-full text-xs">
  1362→                    Edit
  1363→                  </Button>
  1364→                  <Button size="sm" variant="ghost" onClick={() => handleDelete(post.id)} className="rounded-full text-xs text-destructive hover:text-destructive">
  1365→                    Delete
  1366→                  </Button>
  1367→                </div>
  1368→              </div>
  1369→            </div>
  1370→          ))}
  1371→        </div>
  1372→      )}
  1373→    </div>
  1374→  )
  1375→}

  1377→/* ============ BLOG POST FORM ============ */
  1378→function BlogPostForm({ initial, onClose, onSaved }: { initial: BlogPostRow | null; onClose: () => void; onSaved: () => void }) {
  1379→  const [title, setTitle] = React.useState(initial?.title || '')
  1380→  const [slug, setSlug] = React.useState(initial?.slug || '')
  1381→  const [excerpt, setExcerpt] = React.useState(initial?.excerpt || '')
  1382→  const [content, setContent] = React.useState(initial?.content || '[{"type":"p","text":""}]')
  1383→  const [category, setCategory] = React.useState(initial?.category || 'Web Design')
  1384→  const [author, setAuthor] = React.useState(initial?.author || 'Preet Kaur')
  1385→  const [authorRole, setAuthorRole] = React.useState(initial?.authorRole || 'Founder & Creative Director')
  1386→  const [authorInitials, setAuthorInitials] = React.useState(initial?.authorInitials || 'PK')
  1387→  const [authorAccent, setAuthorAccent] = React.useState(initial?.authorAccent || 'from-orange-500 to-pink-500')
  1388→  const [imageUrl, setImageUrl] = React.useState(initial?.imageUrl || '')
  1389→  const [featured, setFeatured] = React.useState(initial?.featured || false)
  1390→  const [status, setStatus] = React.useState(initial?.status || 'published')
  1391→  const [saving, setSaving] = React.useState(false)
  1392→  const [contentType, setContentType] = React.useState<'json' | 'plain'>('json')
  1393→
  1394→  // Parse content for editing
  1395→  React.useEffect(() => {
  1396→    try {
  1397→      const parsed = JSON.parse(content)
  1398→      if (Array.isArray(parsed)) {
  1399→        setContentType('json')
  1400→      }
  1401→    } catch {
  1402→      setContentType('plain')
  1403→    }
  1404→  }, [])
  1405→
  1406→  const generateSlug = (title: string) => {
  1407→    return title.toLowerCase().replace(/[^a-z0-9]+/g, '-').replace(/^-|-$/g, '')
  1408→  }
  1409→
  1410→  const handleTitleChange = (val: string) => {
  1411→    setTitle(val)
  1412→    if (!initial) setSlug(generateSlug(val))
  1413→  }
  1414→
  1415→  const save = async () => {
  1416→    if (!title || !slug) {
  1417→      toast.error('Title and slug are required')
  1418→      return
  1419→    }
  1420→    setSaving(true)
  1421→    try {
  1422→      const method = initial ? 'PUT' : 'POST'
  1423→      const body = { id: initial?.id, title, slug, excerpt, content, category, author, authorRole, authorInitials, authorAccent, imageUrl, featured, status }
  1424→      const res = await fetch('/api/admin/blog', {
  1425→        method,
  1426→        headers: { 'Content-Type': 'application/json' },
  1427→        body: JSON.stringify(body),
  1428→      })
  1429→      const data = await res.json()
  1430→      if (!data.ok) throw new Error(data.error || 'Save failed')
  1431→      toast.success(initial ? 'Post updated!' : 'Post created!')
  1432→      onSaved()
  1433→    } catch (err) {
  1434→      toast.error(err instanceof Error ? err.message : 'Save failed')
  1435→    } finally {
  1436→      setSaving(false)
  1437→    }
  1438→  }
  1439→
  1440→  const parseContentToPlainText = () => {
  1441→    try {
  1442→      const blocks = JSON.parse(content)
  1443→      return blocks.map((b: { text?: string }) => b.text || '').join('\n\n')
  1444→    } catch {
  1445→      return content
  1446→    }
  1447→  }
  1448→
  1449→  const plainToContent = (plain: string) => {
  1450→    const paragraphs = plain.split(/\n\n+/).filter(Boolean)
  1451→    const blocks = paragraphs.map((text) => ({ type: 'p', text: text.trim() }))
  1452→    return JSON.stringify(blocks)
  1453→  }
  1454→
  1455→  return (
  1456→    <div className="rounded-2xl border border-border/60 bg-card p-5 space-y-4">
  1457→      <div className="flex items-center justify-between">
  1458→        <h3 className="font-display text-lg font-bold">{initial ? 'Edit Post' : 'New Blog Post'}</h3>
  1459→        <Button size="sm" variant="ghost" onClick={onClose} className="rounded-full">✕</Button>
  1460→      </div>
  1461→
  1462→      <div className="grid gap-4 sm:grid-cols-2">
  1463→        <div>
  1464→          <label className="mb-1 block text-xs font-semibold">Title *</label>
  1465→          <Input value={title} onChange={(e) => handleTitleChange(e.target.value)} className="bg-muted/30" placeholder="How to scale your ecommerce store..." />
  1466→        </div>
  1467→        <div>
  1468→          <label className="mb-1 block text-xs font-semibold">Slug *</label>
  1469→          <Input value={slug} onChange={(e) => setSlug(e.target.value)} className="bg-muted/30 font-mono text-xs" placeholder="how-to-scale-ecommerce" />
  1470→        </div>
  1471→      </div>
  1472→
  1473→      <div>
  1474→        <label className="mb-1 block text-xs font-semibold">Excerpt (summary shown in card)</label>

  1480→          <label className="text-xs font-semibold">Content</label>
  1481→          <div className="flex gap-1">
  1482→            <button onClick={() => { setContent(plainToContent(parseContentToPlainText())); setContentType('plain') }} className={cn('rounded px-2 py-0.5 text-[10px] font-bold', contentType === 'plain' ? 'bg-brand-gradient text-white' : 'bg-muted text-muted-foreground')}>
  1483→              Plain text
  1484→            </button>
  1485→            <button onClick={() => setContentType('json')} className={cn('rounded px-2 py-0.5 text-[10px] font-bold', contentType === 'json' ? 'bg-brand-gradient text-white' : 'bg-muted text-muted-foreground')}>
  1486→              JSON blocks
  1487→            </button>
  1488→          </div>
  1489→        </div>
  1490→        {contentType === 'plain' ? (
  1491→          <Textarea
  1492→            value={parseContentToPlainText()}
  1493→            onChange={(e) => setContent(plainToContent(e.target.value))}
  1494→            rows={10}
  1495→            className="bg-muted/30 font-mono text-xs"
  1496→            placeholder="Write your article here. Separate paragraphs with a blank line."
  1497→          />
  1498→        ) : (
  1499→          <Textarea
  1500→            value={content}
  1501→            onChange={(e) => setContent(e.target.value)}
  1502→            rows={10}
  1503→            className="bg-muted/30 font-mono text-xs"
  1504→            placeholder='[{"type":"p","text":"Your paragraph"},{"type":"h2","text":"Heading"},{"type":"ul","items":["Item 1","Item 2"]}]'
  1505→          />
  1506→        )}
  1507→        <p className="mt-1 text-[10px] text-muted-foreground">
  1508→          {contentType === 'plain' ? 'Separate paragraphs with blank lines. Each becomes a <p> block.' : 'JSON array of content blocks: {type, text} or {type:"ul", items:[]}'}
  1509→        </p>
  1510→      </div>
  1511→
  1512→      <div className="grid gap-4 sm:grid-cols-3">
  1513→        <div>
  1514→          <label className="mb-1 block text-xs font-semibold">Category</label>

  1516→            <option>Web Design</option><option>AI</option><option>SEO</option><option>Ecommerce</option><option>Growth</option>
  1517→          </select>

  1563→  )
